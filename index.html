let currentScale = 1; let isBarcodeZoomed = false; // 바코드 원클릭 확대 토글 function toggleBarcodeZoom() { const img = document.querySelector('#modalImage'); // 모달 이미지 태그 id const btn = document.querySelector('#btnBarcodeZoom'); if (!isBarcodeZoomed) { // 2.5배 확대 및 바코드가 있는 중간-하단 쪽으로 줌 포커스 currentScale = 2.5; img.style.transform = 'scale(2.5) translateY(-10%)'; btn.innerText = '↺ 원래 크기로 보기'; isBarcodeZoomed = true; } else { currentScale = 1; img.style.transform = 'scale(1) translateY(0)'; btn.innerText = '🔍 바코드 집중 확대 (2.5x)'; isBarcodeZoomed = false; } } // 모달 닫을 때 줌 상태 초기화 function resetZoom() { currentScale = 1; isBarcodeZoomed = false; const img = document.querySelector('#modalImage'); if (img) img.style.transform = 'scale(1) translateY(0)'; }
